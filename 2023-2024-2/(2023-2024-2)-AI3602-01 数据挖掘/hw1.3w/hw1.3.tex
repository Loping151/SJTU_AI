\section*{Question 1}
\begin{align*}
	% jaccard distance
	J(A, B) = \frac{|A \cap B|}{|A \cup B|}
\end{align*}

For ant arbitrarily fixed subset $S$, the probability of picking $k$ elements from $S$ is

\begin{align*}
	P(|S \cap T| = k) = \frac{C^m_k C^{n-m}_{m-k}}{C^n_m},
\end{align*}
and for each possible k, the jaccard distance is $\frac{k}{2m - k}$.

So the expected similarity is
\begin{align*}
	E[J(S, T)] & = \sum_{k=0}^{m} \frac{C^m_k C^{n-m}_{m-k}}{C^n_m} \cdot \frac{k}{2m - k}.
\end{align*}
