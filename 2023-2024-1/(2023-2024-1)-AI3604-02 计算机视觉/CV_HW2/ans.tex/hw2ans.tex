	In 2D cases, the transformation is:
	
	\begin{equation}
		Z_c\left[\begin{array}{l}
			u \\
			v \\
			1
		\end{array}\right]=\left[\begin{array}{ccc}
			\alpha & \lambda & u_0 \\
			0 & \beta & v_0 \\
			0 & 0 & 1
		\end{array}\right]\left[\begin{array}{lll}
			r_{11} & r_{12} & t_1 \\
			r_{21} & r_{22} & t_2 \\
			r_{31} & r_{32} & t_3
		\end{array}\right]\left[\begin{array}{c}
			x \\
			y \\
			1
		\end{array}\right]=\mathbf{K}\left[\begin{array}{lll}
			\mathbf{r}_1 & \mathbf{r}_2 & \mathbf{t}
		\end{array}\right]\left[\begin{array}{c}
			x \\
			y \\
			1
		\end{array}\right],
	\end{equation}
	
	where $Z_c$ is a scaling factor, $K$ is the demanded intrinsic matrix. $(u, v)$ are Image points and $(x, y)$ are world points. Since we got dozens of sets of points on each plane, on each plane we solve 
	
	\begin{equation}
		\mathbf H = \mathbf{K}\left[\begin{array}{lll}
			\mathbf{r}_1 & \mathbf{r}_2 & \mathbf{t}
		\end{array}\right]
	\end{equation}

	using least square method. Now let
	
	\begin{equation}
		\mathbf{B}=\mathbf{K}^{-T} \mathbf{K}^{-1} \equiv\left[\begin{array}{lll}
			B_{11} & B_{12} & B_{13} \\
			B_{12} & B_{22} & B_{23} \\
			B_{13} & B_{23} & B_{33}
		\end{array}\right]
	\end{equation}

	because easily
	
	\begin{equation}
	\begin{aligned}
		& \mathbf{r}_1=\mathbf{K}^{-1} \mathbf{h}_1 \\
		& \mathbf{r}_2=\mathbf{K}^{-1} \mathbf{h}_2
	\end{aligned}
	\end{equation}
	\begin{equation}
	\begin{gathered}
		\mathbf{h}_1^T \mathbf{K}^{-T} \mathbf{K}^{-1} \mathbf{h}_2=0 \\
		\mathbf{h}_1^T \mathbf{K}^{-T} \mathbf{K}^{-1} \mathbf{h}_1=\mathbf{h}_2^T \mathbf{K}^{-T} \mathbf{K}^{-1} \mathbf{h}_2=1
	\end{gathered}
	\end{equation}

	$B$ is symmetric. Now we have 
	
	\begin{equation}
		\begin{aligned}
			\mathbf{h}_i^T \mathbf{B} \mathbf{h}_j = & \left[\begin{array}{lll}
				\mathbf{h}_{1 i} & \mathbf{h}_{2 i} & \mathbf{h}_{3 i}
			\end{array}\right]\left[\begin{array}{lll}
				B_{11} & B_{12} & B_{13} \\
				B_{12} & B_{22} & B_{23} \\
				B_{13} & B_{23} & B_{33}
			\end{array}\right]\left[\begin{array}{c}
				\mathbf{h}_{1 j} \\
				\mathbf{h}_{2 j} \\
				\mathbf{h}_{3 j}
			\end{array}\right] \\
		\stackrel{\triangle}{=}
		& \mathbf v_{ij}^T\left[\begin{array}{l}
				B_{11} \\
				B_{12} \\
				B_{22} \\
				B_{13} \\
				B_{23} \\
				B_{33}
			\end{array}\right]
		=\begin{cases}
		0, i = 1, j = 2\\
		1, i=j=1,2
	\end{cases}
		\end{aligned}
	\end{equation}
	
	\begin{equation}
	\left[\begin{array}{c}
		\mathbf{v}_{12}^T \\
		\mathbf{v}_{11}^T-\mathbf{v}_{22}^T
	\end{array}\right] \left[\begin{array}{l}
	B_{11} \\
	B_{12} \\
	B_{22} \\
	B_{13} \\
	B_{23} \\
	B_{33}
\end{array}\right]=0
	\end{equation}

	Again we use least square to solve $\mathbf B$ and finally use cholesky to solve $\mathbf K$.

	\begin{equation}
		\begin{aligned}
			E(\mathbf{A}, \mathbf{T})&=\sum_{i=1}^N\left\|\mathbf{Y}_i-\mathbf{A} \mathbf{X}_i-\mathbf{T} \right\|_2^2\\
			&=\sum_{i=1}^N\left\|\mathbf{Y}_i\right\|_2^2 - 2 \mathbf{Y}_i^T(\mathbf{A} \mathbf{X}_i+\mathbf{T}) + \left\|(\mathbf{A} \mathbf{X}_i+\mathbf{T})\right\|_2^2
		\end{aligned}
	\end{equation}
	
	For each item in $\sum$:
	
	\begin{equation}
		\begin{cases}
			\frac{\partial E}{\partial \mathbf A}=-2\mathbf Y_i\mathbf X_i^T+2(\mathbf{A} \mathbf{X}_i+\mathbf{T})\mathbf X_i^T\\
			\frac{\partial E}{\partial \mathbf T}=-2\mathbf Y_i+2(\mathbf{A} \mathbf{X}_i+\mathbf{T})
		\end{cases}
	\end{equation}
	
	To find maximum, we let $\frac{\partial E}{\partial \mathbf A}=0$ and $\frac{\partial E}{\partial \mathbf T}=0$.

	\end{equation}
